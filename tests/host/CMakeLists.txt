# PC-side unit tests: the real core, variant and library sources compiled for
# the PC against a fake pico-sdk (fake_sdk/), with simulated on-board chips
# (models/). Configured from the repo root:
#
#   cmake --workflow --preset host-tests
#
# Guide: docs/maintenance/host-tests.html
set(PLATFORM_DIR "${PHYDUINO_PLATFORM_DIR}")
set(BOARD_VARIANT phyduino_pico)

# --- Fake SDK headers ------------------------------------------------------------------
# Every SDK header the tested code includes, each forwarding to fake_sdk.h.
# Code that includes a header not listed here fails to compile: add it here and
# implement what it needs in fake_sdk.h / fake_sdk.cpp.
set(FAKE_SDK_HEADERS
    pico.h
    pico/critical_section.h
    pico/multicore.h
    pico/time.h
    pico/i2c_slave.h
    pico/util/queue.h
    hardware/adc.h
    hardware/clocks.h
    hardware/dma.h
    hardware/gpio.h
    hardware/i2c.h
    hardware/irq.h
    hardware/pwm.h
    hardware/spi.h
    hardware/sync.h
    hardware/uart.h
)
set(FAKE_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/fake_sdk_include")
foreach(header IN LISTS FAKE_SDK_HEADERS)
    file(CONFIGURE OUTPUT "${FAKE_INCLUDE_DIR}/${header}"
         CONTENT "// Generated by tests/host/CMakeLists.txt\n#pragma once\n#include \"fake_sdk.h\"\n")
endforeach()

# --- Code under test --------------------------------------------------------------------
# Everything in the core, variant and libraries, except files that need the
# real USB stack (TinyUSB, ArduinoCore-API PluggableUSB), the linker's heap
# symbols (heap.cpp) or are the program entry point. (freertos.cpp compiles to
# nothing here: the tests are the no-OS build.)
file(GLOB_RECURSE CODE_UNDER_TEST CONFIGURE_DEPENDS
    "${PLATFORM_DIR}/cores/phyduino/*.cpp"
    "${PLATFORM_DIR}/variants/${BOARD_VARIANT}/*.cpp"
    "${PLATFORM_DIR}/libraries/*/src/*.cpp")
list(FILTER CODE_UNDER_TEST EXCLUDE REGEX "/cores/phyduino/(main\\.cpp|heap\\.cpp|SerialUSB\\.cpp|usb/|api/PluggableUSB\\.cpp)")

# Same Arduino defines as the board build (ARDUINO_*, F_CPU, USB_*), from platform.txt/boards.txt.
include("${CMAKE_SOURCE_DIR}/cmake/arduino_properties.cmake")
arduino_load_properties("${PLATFORM_DIR}" "${BOARD_VARIANT}")
arduino_set_property(runtime.ide.version "10607")
arduino_set_property(build.arch "RP2")
arduino_property_args(ARDUINO_DEFINES compiler.arduino.defines build.extra_flags)

file(GLOB TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/test_*.cpp")
file(GLOB LIBRARY_SRC_DIRS LIST_DIRECTORIES true "${PLATFORM_DIR}/libraries/*/src")

add_executable(host_tests
    ${TEST_SOURCES}
    ${CODE_UNDER_TEST}
    fake_sdk/fake_sdk.cpp
    models/chip_models.cpp)
target_compile_features(host_tests PRIVATE cxx_std_17)
target_compile_options(host_tests PRIVATE ${ARDUINO_DEFINES})
# ArduinoCore-API's own PC-build switch: skips its atexit() declaration, which
# clashes with glibc's noexcept one on Linux.
target_compile_definitions(host_tests PRIVATE HOST)
target_include_directories(host_tests PRIVATE
    "${FAKE_INCLUDE_DIR}"
    "${CMAKE_CURRENT_LIST_DIR}/fake_sdk"
    "${CMAKE_CURRENT_LIST_DIR}/models"
    "${PLATFORM_DIR}/cores/phyduino"
    "${PLATFORM_DIR}/variants/${BOARD_VARIANT}"
    ${LIBRARY_SRC_DIRS})
target_include_directories(host_tests SYSTEM PRIVATE "${CMAKE_SOURCE_DIR}/extern/doctest/doctest")

# Strict warnings, except vendored ArduinoCore-API.
set(strict -Wall -Wextra)
if(PHYDUINO_WERROR)
    list(APPEND strict -Werror)
endif()
foreach(src IN LISTS TEST_SOURCES CODE_UNDER_TEST)
    if(src MATCHES "/api/")
        set_source_files_properties("${src}" PROPERTIES COMPILE_OPTIONS -w)
    else()
        set_source_files_properties("${src}" PROPERTIES COMPILE_OPTIONS "${strict}")
    endif()
endforeach()

# MinGW: link the C++ runtime statically, so the exe never picks up another
# libstdc++ DLL found first on PATH (fails with 0xc0000139).
if(MINGW)
    target_link_options(host_tests PRIVATE -static)
endif()

add_test(NAME host_tests COMMAND host_tests)
